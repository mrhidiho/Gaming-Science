cmake_minimum_required(VERSION 3.20)
project(GameEngineeringPackage06 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GE_SANITIZERS "Enable Address/Undefined sanitizers with GCC/Clang" OFF)
add_executable(cpu_renderer main.cpp)
add_executable(renderer_tests tests.cpp)
foreach(target cpu_renderer renderer_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    if(GE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME cpu_renderer_contracts COMMAND renderer_tests)
