cmake_minimum_required(VERSION 3.16)
project(volume6_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(audio_harness audio_harness.cpp)
if(MSVC)
 target_compile_options(audio_harness PRIVATE /W4)
else()
 target_compile_options(audio_harness PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME offline_audio COMMAND audio_harness)
