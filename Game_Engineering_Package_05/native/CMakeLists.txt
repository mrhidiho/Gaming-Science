cmake_minimum_required(VERSION 3.20)
project(GameEngineeringSystems LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STUDENT_EXERCISES "Test unfinished student headers instead of reference headers" OFF)
set(SANITIZER "none" CACHE STRING "none, address, or thread (separate builds)")
find_package(Threads REQUIRED)
add_library(course_options INTERFACE)
target_link_libraries(course_options INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(course_options INTERFACE /W4 /permissive-)
else()
  target_compile_options(course_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(NOT SANITIZER STREQUAL "none")
  if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "This teaching sanitizer configuration needs GCC or Clang")
  endif()
  if(SANITIZER STREQUAL "address")
    set(SAN_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(SANITIZER STREQUAL "thread")
    set(SAN_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
  else()
    message(FATAL_ERROR "SANITIZER must be none, address or thread")
  endif()
  target_compile_options(course_options INTERFACE ${SAN_FLAGS})
  target_link_options(course_options INTERFACE ${SAN_FLAGS})
endif()
add_executable(contract_tests tests/contracts.cpp)
target_link_libraries(contract_tests PRIVATE course_options)
if(STUDENT_EXERCISES)
  target_include_directories(contract_tests PRIVATE student)
else()
  target_include_directories(contract_tests PRIVATE reference)
endif()
enable_testing()
foreach(id 16.1 16.2 16.3 16.4 18.1 18.2 18.3 18.4)
  add_test(NAME lab_${id} COMMAND contract_tests ${id})
  set_tests_properties(lab_${id} PROPERTIES TIMEOUT 30)
endforeach()
foreach(id 16_1 16_2 16_3 16_4 18_1 18_2 18_3 18_4)
  add_executable(demo_${id} examples/lesson_${id}.cpp)
  target_include_directories(demo_${id} PRIVATE reference)
  target_link_libraries(demo_${id} PRIVATE course_options)
endforeach()
add_executable(native_worker worker.cpp)
target_link_libraries(native_worker PRIVATE course_options)
add_executable(benchmark benchmark.cpp kernels.cpp)
target_link_libraries(benchmark PRIVATE course_options)

add_executable(graph_tests tests/graph.cpp)
target_include_directories(graph_tests PRIVATE reference)
target_link_libraries(graph_tests PRIVATE course_options)
add_test(NAME graph_integration COMMAND graph_tests)
set_tests_properties(graph_integration PROPERTIES TIMEOUT 30)
foreach(name locality_benchmark false_sharing graph_benchmark)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE course_options)
  target_include_directories(${name} PRIVATE reference)
endforeach()
