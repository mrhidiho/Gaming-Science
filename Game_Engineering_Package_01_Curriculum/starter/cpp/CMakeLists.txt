cmake_minimum_required(VERSION 3.16)
project(GameEngineeringStarter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(motion_demo main.cpp)
add_executable(motion_tests tests.cpp)
foreach(target motion_demo motion_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME motion_contract COMMAND motion_tests)
