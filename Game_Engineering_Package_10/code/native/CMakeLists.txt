cmake_minimum_required(VERSION 3.16)
project(volume09_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable Clang/GCC address and undefined behavior sanitizers" OFF)
add_executable(v09_native main.cpp)
if(MSVC)
  target_compile_options(v09_native PRIVATE /W4)
else()
  target_compile_options(v09_native PRIVATE -Wall -Wextra -Wpedantic)
  if(ENABLE_SANITIZERS)
    target_compile_options(v09_native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(v09_native PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_test(NAME native_semantics COMMAND v09_native)
