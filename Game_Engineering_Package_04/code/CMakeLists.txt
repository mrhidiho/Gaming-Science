cmake_minimum_required(VERSION 3.20)
project(GameEngineeringPackage04 LANGUAGES C CXX)
include(CTest)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(P4_SANITIZE "Enable AddressSanitizer and UBSan with GCC/Clang" OFF)
option(P4_WARNINGS_AS_ERRORS "Fail compilation on warnings" ON)
option(P4_BUILD_STUDENT "Build/register incomplete student exercises too" OFF)
add_library(p4_options INTERFACE)
target_include_directories(p4_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/c_bridge)
if(MSVC)
  target_compile_options(p4_options INTERFACE /W4)
  if(P4_WARNINGS_AS_ERRORS)
    target_compile_options(p4_options INTERFACE /WX)
  endif()
else()
  target_compile_options(p4_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(P4_WARNINGS_AS_ERRORS)
    target_compile_options(p4_options INTERFACE -Werror)
  endif()
endif()
if(P4_SANITIZE)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT MSVC)
    target_compile_options(p4_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(p4_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  else()
    message(FATAL_ERROR "P4_SANITIZE supports the tested GCC/Clang flag family; use your platform's documented tools.")
  endif()
endif()
add_library(ge_c_api STATIC c_bridge/ge_c_api.c)
target_link_libraries(ge_c_api PRIVATE p4_options)
foreach(module RANGE 11 15)
  foreach(lesson RANGE 1 4)
    set(name lesson_${module}_${lesson})
    add_executable(${name} reference/${name}.cpp)
    target_link_libraries(${name} PRIVATE p4_options)
    if(name STREQUAL "lesson_11_1")
      target_sources(${name} PRIVATE reference/tick.cpp)
    endif()
    if(name STREQUAL "lesson_15_1")
      target_link_libraries(${name} PRIVATE ge_c_api)
    endif()
    if(BUILD_TESTING)
      add_test(NAME reference_${module}_${lesson} COMMAND ${name})
      set_tests_properties(reference_${module}_${lesson} PROPERTIES LABELS "reference;module_${module}")
    endif()
    if(P4_BUILD_STUDENT)
      add_executable(student_${module}_${lesson} starter/${name}.cpp)
      target_link_libraries(student_${module}_${lesson} PRIVATE p4_options ge_c_api)
      if(BUILD_TESTING)
        add_test(NAME student_${module}_${lesson} COMMAND student_${module}_${lesson})
        set_tests_properties(student_${module}_${lesson} PROPERTIES LABELS "student")
      endif()
    endif()
  endforeach()
endforeach()
foreach(pair "c01_basics.c" "c02_ownership.cpp" "c03_layout.c" "c04_errors.cpp")
  get_filename_component(name ${pair} NAME_WE)
  add_executable(${name} c_bridge/${pair})
  target_link_libraries(${name} PRIVATE p4_options ge_c_api)
  if(BUILD_TESTING)
    add_test(NAME reference_${name} COMMAND ${name})
    set_tests_properties(reference_${name} PROPERTIES LABELS "reference;c_bridge")
  endif()
endforeach()
add_executable(layout_benchmark reference/layout_benchmark.cpp)
target_link_libraries(layout_benchmark PRIVATE p4_options)
