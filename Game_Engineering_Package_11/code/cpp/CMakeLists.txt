cmake_minimum_required(VERSION 3.16)
project(volume10_publication LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(publication publication.cpp)
target_compile_features(publication PRIVATE cxx_std_17)
target_link_libraries(publication PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(publication PRIVATE /W4 /WX)
else()
  target_compile_options(publication PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME publication_smoke COMMAND publication 200)
add_test(NAME rejects_zero COMMAND publication 0)
set_tests_properties(rejects_zero PROPERTIES WILL_FAIL TRUE)
add_test(NAME rejects_text COMMAND publication abc)
set_tests_properties(rejects_text PROPERTIES WILL_FAIL TRUE)
set_tests_properties(publication_smoke PROPERTIES TIMEOUT 30)
